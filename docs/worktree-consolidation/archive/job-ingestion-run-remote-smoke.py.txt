def tabs(source):
    result = subprocess.run(['opencli', '--profile', profile, 'browser', f'imx-jobs-{source}', 'tab', 'list'],
                            capture_output=True, text=True, check=True, timeout=30)
    value = json.loads(result.stdout)
    if protected in result.stdout:
        raise RuntimeError('Protected assessment tab is unexpectedly in a jobs-owned session; stopped')
    return value

    audit['before'][source] = tabs(source)

class SmokeTransport(OpenCliTransport):
    def __init__(self):
        super().__init__(profile=profile, window='background', timeout_s=45)
        self.searches = {}
        self.details = {}

    def open(self, session, url):
        ref = known_source_ref(url)
        counts = self.details if ref is not None else self.searches
        limit = 1
        if counts.get(session, 0) >= limit:
            raise AccessProblem(SourceSearchState.PARTIAL, 'Bounded smoke stopped before another search/detail page; no further source coverage attempted')
        counts[session] = counts.get(session, 0) + 1
        super().open(session, url)

    def _run(self, session, args, *, timeout_s=None):
        assert session.startswith('imx-jobs-') and protected not in args
        entry = {'session': session, 'command': args[0]}
        if args[0] == 'open':
            entry['url'] = args[1]
        elif args[0] == 'click':
            assert args[1] == '[data-share-url]'
            entry['selector'] = args[1]
        audit['commands'].append(entry)
        return super()._run(session, args, timeout_s=timeout_s)

    def evaluate(self, session, script):
        payload = super().evaluate(session, script)
        if isinstance(payload, dict):
            page = payload.get('page') or {}
            audit['source_pages'].append({'session': session, 'url': page.get('url'), 'title': page.get('title'),
                'signals': page.get('signals'), 'cards_seen': len(payload.get('cards') or payload.get('items') or []),
                'detail_heading': (payload.get('active') or {}).get('heading'),
                'detail_doc_id': (payload.get('active') or {}).get('encoded_docid')})
        return payload
